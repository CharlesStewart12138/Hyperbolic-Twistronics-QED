\section{Weighted resolvent locality}
\label{sec:r7-resolvent}

For subsets $X,Y$ of the two-layer site set, let $d(X,Y)$ be the product
metric with zero layer cost and hyperbolic in-plane distance.  Write $\chi_X$
for the corresponding projection.

\begin{theorem}[Combes--Thomas bound with volume growth included]
\label{thm:r7-ct}
Let $z=E+i\eta$, $\eta\ne0$, and choose
$0<q<\mu-\kappa$ with $\Phi(q)<|\eta|$, where $\Phi$ is
\eqref{eq:r7-Phi}.  Then
\begin{equation}
 \|\chi_X(H(\theta)-z)^{-1}\chi_Y\|
 \le\frac{e^{-qd(X,Y)}}{|\eta|-\Phi(q)}.
 \label{eq:r7-ct}
\end{equation}
The constants are uniform in $\theta$.
\end{theorem}

\begin{proof}
For a one-Lipschitz truncated distance $f$, Schur's test gives
\begin{equation}
 \|e^{qf}He^{-qf}-H\|\le
 \sup_x\sum_y|H_{xy}|(e^{qd(x,y)}-1)\le\Phi(q).
\end{equation}
The last inequality is exactly the intralayer nearest-neighbour term plus
the difference of the two convergent series $S(\mu-q)-S(\mu)$.
The resolvent of the conjugated operator exists by a Neumann series and has
norm at most $(|\eta|-\Phi(q))^{-1}$.  Letting the truncation of $f$ tend to
$d(\,cdot\,,X)$ proves \eqref{eq:r7-ct}.
\end{proof}

\begin{corollary}[Finite-ball resolvent error]
\label{cor:r7-res-trunc}
Let $X\subset B_\ell(o)$, $L>\ell$, $b=L-\ell$, and let
$H_L=P_LHP_L$ act on $P_L\mathscr H$.  Under the hypotheses of
Theorem~\ref{thm:r7-ct},
\begin{equation}
 \|\chi_X[(z-H)^{-1}-(z-H_L)^{-1}]\chi_X\|
 \le \varepsilon_{\rm res}(b,\eta,q),
 \quad
 \varepsilon_{\rm res}=
 \frac{2M e^{-qb}}{|\eta|(|\eta|-\Phi(q))}.
 \label{eq:r7-res-trunc}
\end{equation}
Here the finite resolvent is extended by zero outside $P_L\mathscr H$.
\end{corollary}

\begin{proof}
Compare $H$ with the block-decoupled
$\widehat H_L=P_LHP_L+(1-P_L)H(1-P_L)$.  Their difference is the two boundary
blocks and has norm at most $2M$.  The resolvent identity, one weighted
off-diagonal estimate from $X$ to the boundary, and the elementary resolvent
bound $|\eta|^{-1}$ give \eqref{eq:r7-res-trunc}.  Compressing the
$P_L$ block gives the stated formula.
\end{proof}

In particular,
\begin{equation}
 |\rho_{x,\eta}(E;H)-\rho_{x,\eta}(E;H_L)|
 \le \pi^{-1}\varepsilon_{\rm res},
\end{equation}
so finite-broadening LDOS is local.  This statement makes no assertion about
the singular limit $\eta\downarrow0$.

For two finite matrices $A,B$ with $\|A\|,\|B\|\le M$,
\begin{align}
 |\langle x|A^n-B^n|x\rangle|&\le nM^{n-1}\|A-B\|,\\
 |\langle x|(z-A)^{-1}-(z-B)^{-1}|x\rangle|
 &\le |\eta|^{-2}\|A-B\|.
 \label{eq:r7-local-observable-lipschitz}
\end{align}
Combining these inequalities with Theorem~\ref{thm:r7-twist} and
Corollary~\ref{cor:r7-res-trunc} proves local twist continuity of moments,
resolvents, and broadened LDOS.  Exact infinite moments can be treated without
a finite matrix: truncate hopping range at $r$, use
\eqref{eq:r7-tail}, observe that an $n$-step path remains in $B_{nr}$, and
obtain
\begin{equation}
 |m_n(x;\theta)-m_n(x;\theta')|
 \le nM^{n-1}\left[2C_{\rm tail}e^{-\gamma r}
 +C_{nr}|\theta-\theta'|\right].
 \label{eq:r7-moment-bound}
\end{equation}
